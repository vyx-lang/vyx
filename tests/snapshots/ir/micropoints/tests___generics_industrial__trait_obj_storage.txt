# micropoints — trait_obj_storage  （codegen path: trait dyn dispatch via Vec-stored vtable fat-ptrs）
# 仅在 O0 baseline 上断言。

# Job trait 的 vtable 类型形态
must_have      %Job_vtable = type \{ ptr \}

# 三个具体 impl 的 vtable 常量
must_have      @Inc_vtable_Job = internal constant %Job_vtable \{ ptr @Inc\.run \}
must_have      @Sq_vtable_Job = internal constant %Job_vtable \{ ptr @Sq\.run \}
must_have      @Neg_vtable_Job = internal constant %Job_vtable \{ ptr @Neg\.run \}

# 间接 dispatch 形态：iface 槽 → vtbl 槽 → fn-ptr
must_have      iface\.vtbl\.slot
must_have      vtbl\.fn\.slot
must_have      iface\.call = call i32 %vtbl\.fn

# 反向锁：不应"反优化"成静态 call（Vec 存储动态 dispatch 必须保持 indirect）
must_not_have  call i32 @Inc\.run\(ptr
must_not_have  call i32 @Sq\.run\(ptr
must_not_have  call i32 @Neg\.run\(ptr

# unwind future-proof 哨兵
must_not_have  (?m)^\s*invoke\s+
must_not_have  cleanuppad
must_not_have  landingpad
must_not_have  @_Znwm
