# micropoints — adv_05_pack_of_10  （codegen path: 10-元 pack 单态化 + reflect/pack 大场景）
# 仅在 O0 baseline 上断言。

# 10 元 pack 单态化函数必须 emit（mangled name 锁形态）
must_have      define i64 @"sum10<i64,i64,i64,i64,i64,i64,i64,i64,i64,i64>"\(

# 10 个参数 alloca （每个 args_N 分配一个 i64 槽）
count          %args_\d+\d* = alloca i64 >= 10

# 9 次累加（10 元 reduce-fold）
count          %add\d* = add i64 >= 9

# main 调用点把 10 个字面量塞进去
must_have      call i64 @"sum10<i64,i64,i64,i64,i64,i64,i64,i64,i64,i64>"\(i64 1, i64 2, i64 3, i64 4, i64 5, i64 6, i64 7, i64 8, i64 9, i64 10\)

# pack-of-N 不能错走可变参数（@printf 是 vararg, sum10 必须是 fixed-arity）
must_not_have  define i64 @"sum10<.*>".*\.\.\.\)

# 不应链入 C++ 异常 runtime
must_not_have  @_Znwm
