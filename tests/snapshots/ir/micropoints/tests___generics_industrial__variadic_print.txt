# micropoints — variadic_print  （codegen path: variadic pack 单态化 + arity 枚举）
# 仅在 O0 baseline 上断言。
# 验证 ...T 在 i64/f64/i32 三种基本类型 + 不同 arity 下都能正确单态化。

# 三档 sum_all/print_many 不同 T+arity 单态化（fixed-arity 而非 vararg）
must_have      define i64 @"sum_all<i64>"\(i64 %args_0, i64 %args_1, i64 %args_2, i64 %args_3, i64 %args_4\)
must_have      define double @"sum_all<f64>"\(double %args_0, double %args_1, double %args_2\)
must_have      define void @"print_many<i32>"\(%__String %label, i32 %args_0, i32 %args_1, i32 %args_2, i32 %args_3\)

# pack 必须不被错走 vararg ...
must_not_have  define i64 @"sum_all<.*>".*\.\.\.\)
must_not_have  define double @"sum_all<.*>".*\.\.\.\)

# main 调用点必须把字面量逐个塞进去（不是数组打包）
must_have      call i64 @"sum_all<i64>"\(i64 1, i64 2, i64 3, i64 4, i64 5\)
must_have      call double @"sum_all<f64>"\(double 1\.500000e\+00, double 2\.500000e\+00, double 3\.000000e\+00\)

# unwind future-proof 哨兵
must_not_have  (?m)^\s*invoke\s+
must_not_have  cleanuppad
must_not_have  landingpad
must_not_have  @_Znwm
