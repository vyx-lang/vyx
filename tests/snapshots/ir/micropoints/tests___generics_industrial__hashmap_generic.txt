# micropoints — hashmap_generic  （codegen path: HashMap<K,V> 跨两种 K 类型的双单态化）
# 仅在 O0 baseline 上断言。

# 两档 HashMap 单态化：K=string,V=i64 与 K=i64,V=string
must_have      %"HashMap<string,i64>" = type
must_have      %"HashMap<i64,string>" = type

# 两档 .new() 必须各自 emit（key/val 类型不同 → 不同 mangled name）
must_have      define %"HashMap<string,i64>" @"HashMap<string,i64>\.new"\(\)
must_have      define %"HashMap<i64,string>" @"HashMap<i64,string>\.new"\(\)

# 关键方法：put / get / size 必须各档分别单态化
count          define void @"HashMap<string,i64>\.put"\( >= 1
count          define void @"HashMap<i64,string>\.put"\( >= 1
count          define %__String @"HashMap<i64,string>\.get"\( >= 1

# 至少 16 个 HashMap.* 单态化函数（两档 × 8 主方法）
count          define .* @"HashMap<[^"]+>\.[\w_]+"\( >= 16

# unwind future-proof 哨兵
must_not_have  (?m)^\s*invoke\s+
must_not_have  cleanuppad
must_not_have  landingpad
must_not_have  @_Znwm
