# micropoints — multi_param_generic  （codegen path: 多参数泛型 <A,B,C> 单态化）
# 仅在 O0 baseline 上断言。
# 关键观测：Triple<A,B,C> 经 pack/rotate 二轮重排后产生 3 个不同结构 layout 的单态化。

# 三档 Triple 单态化结构体类型必须各自 emit
must_have      %"Triple<i32,string,bool>" = type
must_have      %"Triple<string,bool,i32>" = type
must_have      %"Triple<bool,i32,string>" = type

# pack<i32,string,bool>(...) 单态化函数
must_have      define %"Triple<i32,string,bool>" @"pack<i32,string,bool>"\(

# 两次 rotate 单态化（参数序不同 → 不同 mangled name）
must_have      define %"Triple<string,bool,i32>" @"rotate<i32,string,bool>"\(
must_have      define %"Triple<bool,i32,string>" @"rotate<string,bool,i32>"\(

# 累计至少 3 个 Triple 类型 + 3 个泛型函数单态化
count          define .* @"(pack|rotate)<.*>"\( >= 3

# 静态 dispatch（无 vtable 间接调用）
must_not_have  vtbl\.fn\.slot

# unwind future-proof 哨兵
must_not_have  (?m)^\s*invoke\s+
must_not_have  cleanuppad
must_not_have  landingpad
must_not_have  personality
must_not_have  @_Znwm
