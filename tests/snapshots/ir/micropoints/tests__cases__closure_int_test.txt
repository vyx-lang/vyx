# micropoints — closure_int_test  （codegen path: closure capture by-value + heap env）
# 仅在 O0 baseline 上断言（O2 inlining/DCE 后不稳定）。

# fat-pointer 类型必须出现（闭包 ABI 形态锁）
must_have      __closure_fat_ptr

# 所有 3 个闭包的 env 都必须通过 malloc 堆分配（不可走栈，否则跨调用作用域会悬挂）
must_have      __closure_\d+\.env\.alloc = call ptr @malloc
count          __closure_\d+\.env\.alloc = call ptr @malloc >= 3

# fat ptr 自身也走 malloc（fn-ptr + env-ptr 一对）
count          __closure_\d+\.fat\.alloc = call ptr @malloc >= 3

# 必须有 3 个内部 closure 函数体被 emit（多行模式）
count          (?m)^define internal i32 @__closure_\d+\(ptr %env >= 3

# 必须不出现 C++ new（确认走 vyxc 自家 malloc 路径）
must_not_have  @_Znwm

# capture 字段必须以 envfield/envptr 形态保留（防止 codegen 漂移把 env 拍平）
must_have      \.envfield
must_have      \.envptr
