# micropoints — nested_dict_vec  （codegen path: Dict<string, Vec<i64>> 嵌套容器单态化）
# 仅在 O0 baseline 上断言。

# Dict<string,Vec<i64>> 复合类型单态化
must_have      %"Dict<string,Vec<i64>>" = type
must_have      %"Vec<i64>" = type

# 关键 Dict 方法的单态化（new / put / get / contains / size 等需各自 emit）
must_have      define %"Dict<string,Vec<i64>>" @"Dict<string,Vec<i64>>\.new"\(\)
must_have      define void @"Dict<string,Vec<i64>>\.put"\(
must_have      define void @"Dict<string,Vec<i64>>\._grow"\(

# 内部辅助方法（rehash/keyhash 等私有 helper 应该出现）
must_have      define i64 @"Dict<string,Vec<i64>>\._hash_key"\(
must_have      define ptr @"Dict<string,Vec<i64>>\._ctrl_ptr"\(

# 嵌套元素 Vec<i64> 的 push/clone/drop 应被单态化
must_have      define void @"Vec<i64>\.push"\(ptr %self, i64 %val\)
must_have      define void @"Vec<i64>\.drop"\(

# 至少 10 个 Dict<string,Vec<i64>>.<method> 单态化函数
count          define .* @"Dict<string,Vec<i64>>\.[\w_]+"\( >= 10

# unwind future-proof 哨兵
must_not_have  (?m)^\s*invoke\s+
must_not_have  cleanuppad
must_not_have  landingpad
must_not_have  @_Znwm
