# micropoints — second_order  （codegen path: HashSet<Vec<i32>> = Dict<Vec<i32>, bool> 二阶嵌套）
# 仅在 O0 baseline 上断言。

# Dict<Vec<i32>, bool> 嵌套类型单态化（key 是 generic-Vec）
must_have      %"Dict<Vec<i32>,bool>" = type
must_have      %"Vec<i32>" = type

# Dict<Vec<i32>,bool> 的 new / put / contains / size 必须单态化
must_have      define %"Dict<Vec<i32>,bool>" @"Dict<Vec<i32>,bool>\.new"\(\)
must_have      define void @"Dict<Vec<i32>,bool>\.put"\(
must_have      define i8 @"Dict<Vec<i32>,bool>\.contains"\(

# 内嵌 Vec<i32>.push 单态化（key 类型自身的 generic 链路）
must_have      define void @"Vec<i32>\.push"\(ptr %self, i32 %val\)

# 复合类型的 hash key 路径（_hash_key 必须在 K=Vec<i32> 上专门 emit）
must_have      define i64 @"Dict<Vec<i32>,bool>\._hash_key"\(

# 至少 12 个 Dict<Vec<i32>,bool>.<method> 单态化（Dict 的 generic 链路在嵌套 K 下不应被裁剪）
count          define .* @"Dict<Vec<i32>,bool>\.[\w_]+"\( >= 12

# unwind future-proof 哨兵
must_not_have  (?m)^\s*invoke\s+
must_not_have  cleanuppad
must_not_have  landingpad
must_not_have  @_Znwm
