# micropoints — box_dyn_trait  （codegen path: trait dyn dispatch via vtable）
# 仅在 O0 baseline 上断言。

# Greeter trait 的 vtable 形状（fat-pointer 双槽）
must_have      %__iface_Greeter = type \{ ptr, ptr \}
must_have      %Greeter_vtable = type \{ ptr \}

# 两个具体 impl 的 vtable 常量必须 emit
must_have      @En_vtable_Greeter = internal constant %Greeter_vtable \{ ptr @En\.greet \}
must_have      @Zh_vtable_Greeter = internal constant %Greeter_vtable \{ ptr @Zh\.greet \}

# greet 方法体（具体实现）必须存在
must_have      define %__String @En\.greet\(ptr %self\)
must_have      define %__String @Zh\.greet\(ptr %self\)

# 动态分派核心：从 iface fat-ptr 取出 vtable 槽 → 取 fn-ptr → 间接 call
must_have      iface\.vtbl\.slot
must_have      vtbl\.fn\.slot
must_have      iface\.call = call %__String %vtbl\.fn

# 必须 emit 至少 2 次 iface.call（main 里 g_en + g_zh 各一次）
count          %iface\.call\d* = call %__String %vtbl\.fn >= 2

# 不应"反优化"成静态分派（两个 impl 的方法不应被直接 call 到）—— 必须走 vtbl.fn
must_not_have  call %__String @En\.greet\(
must_not_have  call %__String @Zh\.greet\(

# 不应链入 C++ 异常 runtime
must_not_have  @_Znwm
