# micropoints — lambda_capture_generic  （codegen path: 泛型闭包 capture + 延迟实例化 / @[lazy]-like 形态）
# 仅在 O0 baseline 上断言。
# 闭包 = 延迟求值的载体；本用例锁住"泛型 lambda 在不同调用点的 fat-ptr ABI 与 env 堆分配形态"，
# 任何 codegen 把 env 拍平到栈、把 fat-ptr 改成 raw-ptr 都将立刻被察觉。

# fat-pointer ABI 形态（不可改）
must_have      %__closure_fat_ptr = type \{ ptr, ptr \}

# closure env 类型 + 堆分配（malloc）路径
must_have      %__closure_\d+\.env = type
must_have      __closure_\d+\.env\.alloc = call ptr @malloc
must_have      __closure_\d+\.fat\.alloc = (?:tail )?call ptr @malloc

# closure body 函数（以 `define internal` + ptr %env 为形态锁）
count          (?m)^define internal .* @__closure_\d+\(ptr %env >= 1

# 调用形态：从 fat-ptr 取 fnptr/envptr 再间接 call —— 必须保留 fat-call 形态
must_have      \.fat\.call = (?:tail |notail |musttail )?call .*%[\w\.]+\.fnptr\(ptr %[\w\.]+\.envptr

# 反向锁：不应回退到 raw 函数指针（无 env 槽）
must_not_have  define i32 @__closure_\d+\(i32 %x\)

# unwind future-proof 哨兵（lazy 形态对 unwind 重构敏感）
must_not_have  (?m)^\s*invoke\s+
must_not_have  cleanuppad
must_not_have  landingpad
must_not_have  personality
must_not_have  @_Znwm
