# micropoints — _p4_early_return_drop  （codegen path: P4 auto-drop on early-return）
# 仅在 O0 baseline 上断言。

# helper() 函数本身必须 emit（auto-drop 注入点）
must_have      define void @helper\(

# 早返回路径上每条 return 之前必须 emit Ref<T>::drop
# main 内 4 个早返回 + helper 内 2 个分支 = 至少 6 次 drop
count          call void @"Ref<i64>\.drop" >= 6

# 早返回前 if-then 块必须存在（验证 control-flow 形态）
must_have      then:

# probe 与 local 两个变量必须各自 emit 一次 drop（auto-drop scope-aware）
count          @"Ref<i64>\.drop"\(ptr %probe1\) >= 1
count          @"Ref<i64>\.drop"\(ptr %local\) >= 1

# 不应漏掉 free（drop 链最终必须落到 @free）
must_have      call void @free
