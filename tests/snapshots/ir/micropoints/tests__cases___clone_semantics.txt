# micropoints — _clone_semantics  （codegen paths: clone 语义 + unwind future-proof）
# 仅在 O0 baseline 上断言。
#
# 本用例兼任「面向未来 unwind 重构」哨兵：当前 panic 路径走
# `__vyx_panic` → `printf` → `exit(1)` → `unreachable`，是 no-unwind 形态。
# 一旦 codegen 切换到 invoke + landingpad/cleanuppad（SEH/Itanium EH），
# 这些 must_not_have 会立刻在 0c 报警，强制评审显式 bless 新基线。

# clone 链必须存在（std.clone.Clone vtable 形态）
must_have      std\.clone\.Clone_vtable
must_have      std\.collections\.String\.clone

# Drop 链必须 emit
must_have      std\.hash\.Drop_vtable

# panic 路径当前形态：call/ret/exit/unreachable，无 invoke/landingpad
must_have      __vyx_panic
must_have      unreachable
must_have      call void @exit\(i32 1\)

# ====== unwind future-proof 哨兵 ======
# unwind 重构会引入下列任一形态——出现即立刻 0c FAIL，要求人工评审。
must_not_have  (?m)^\s*invoke\s+
must_not_have  cleanuppad
must_not_have  landingpad
must_not_have  personality
must_not_have  cleanupret
must_not_have  catchpad
must_not_have  catchswitch

# C++ 运行时不应被链入
must_not_have  @_Znwm
must_not_have  @__cxa_
