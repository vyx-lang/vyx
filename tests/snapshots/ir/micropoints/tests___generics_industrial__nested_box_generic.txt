# micropoints — nested_box_generic  （codegen path: Box<Box<T>> 嵌套 smart-ptr 单态化）
# 仅在 O0 baseline 上断言。

# 内层 Box<i32> 全套方法单态化
must_have      define ptr @"Box<i32>\.new"\(i32 %val\)
must_have      define i32 @"Box<i32>\.deref"\(ptr %self\)
must_have      define ptr @"Box<i32>\.clone"\(ptr %self\)
must_have      define void @"Box<i32>\.drop"\(ptr %self\)

# 外层 Box<Box<T>> 的变量 ABI 是 raw payload pointer；链式 deref
# 直接从 payload slot 取内层 Box handle，不再调用包装方法。
must_have      %box\.inferred\.deref = load ptr, ptr %box\.ptr
must_not_have  call ptr @"Box<Box<i32>>\.deref"\(
must_not_have  call ptr @"Box<Box<string>>\.deref"\(

# 至少 8 个 Box<*> 函数定义/声明（new/get/deref/set/clone/drop × 多 T）
count          (?:define|declare) .* @"Box<[^"]+>\.\w+"\( >= 8

# malloc 路径（Box.new 内部必须 emit malloc，未走 C++ new）
must_have      call ptr @malloc
must_not_have  @_Znwm

# unwind future-proof 哨兵
must_not_have  (?m)^\s*invoke\s+
must_not_have  cleanuppad
must_not_have  landingpad
must_not_have  @__cxa_
