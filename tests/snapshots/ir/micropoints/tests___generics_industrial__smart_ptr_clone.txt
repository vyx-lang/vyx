# micropoints — smart_ptr_clone  （codegen paths: generic 单态化 method-call + panic 路径）
# 仅在 O0 baseline 上断言。

# Vec<T>::push 三档单态化（i64 / string / Node）—— 非 vararg 形态
must_have      define void @"Vec<i64>\.push"\(ptr %self, i64 %val\)
must_have      define void @"Vec<string>\.push"\(ptr %self, %__String %val\)
must_have      define void @"Vec<Node>\.push"\(ptr %self, %Node %val\)
count          define void @"Vec<.*>\.push"\( >= 3

# Node.clone（用户类型 derive Clone）必须 emit
must_have      define %Node @Node\.clone\(ptr %self\)

# Vec<T>::push_unchecked / Vec<T>::drop / Vec<T>::get 三档单态化也要存在（generic 全套链路）
count          define .* @"Vec<.*>\.push_unchecked"\( >= 3
count          define .* @"Vec<.*>\.drop"\( >= 3

# panic 调用点（用于后续 unwind 重构可推列）
must_have      define internal void @__vyx_panic\(
must_have      call void @exit\(i32 1\)
must_have      unreachable

# panic 当前是 no-unwind 形态（重构后会换 invoke + landingpad，本断言会立刻报警）
must_not_have  (?m)^\s*invoke\s+
must_not_have  cleanuppad
must_not_have  landingpad

# 不应链入 C++ 异常 runtime
must_not_have  @_Znwm
must_not_have  @__cxa_
